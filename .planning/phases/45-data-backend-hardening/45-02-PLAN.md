**For the catch block (network failure fallback):**
- In the existing `catch` block of `loadMessageHistory()`, before or after setting the error, add:
  ```swift
  // Offline fallback: load cached messages (DATA-03)
  if let sessionId = sessionId {
      let cached = await CacheService.shared.getCachedMessages(forSession: sessionId)
      if !cached.isEmpty {
          messages = cached.map { message in
              ChatMessage(
                  id: message.id,
                  isUser: message.role == .user,
                  text: message.content,
                  toolCalls: parseToolCalls(from: message.toolCalls),
                  timestamp: message.createdAt,
                  isFromHistory: true
              )
          }
          AppLogger.shared.info("Loaded \(cached.count) cached messages for session \(sessionId)", category: "chat")
      }
  }
  ```

**IMPORTANT:** `CacheService.shared.cacheMessages` expects `[Message]` (ILSShared model), not `[ChatMessage]` (view model). Cache the raw API response items (`finalItems`), not the mapped `loadedMessages`.

```swift
.onChange(of: inputText) { _, newValue in
    draftPersistTask?.cancel()
    draftPersistTask = Task {
        try? await Task.sleep(nanoseconds: 500_000_000) // 500ms debounce
        guard !Task.isCancelled else { return }
        let key = "chatDraft_\(session.id.uuidString)"
        if newValue.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
            UserDefaults.standard.removeObject(forKey: key)
        } else {
            UserDefaults.standard.set(newValue, forKey: key)
        }
    }
}
```
