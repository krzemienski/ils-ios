4. Add the main merge function `readEffectiveConfig() -> EffectiveConfig`. The merge strategy:

- Read all 4 scopes: `user` (try?), `project` (try?), `local` (try?), `managed` (readManagedConfig). Missing files = empty config, not errors.
- Precedence order (highest last): user, project, local, managed
- For each of ClaudeConfig's 12 optional properties, iterate through scopes in precedence order. The last scope with a non-nil value wins.
- Build a `ConfigOverride` for each key that has at least one non-nil value across scopes.
- Use `String(describing:)` to serialize winning values to String for the `winningValue` field. For complex types (PermissionsConfig, HooksConfig, etc.), use JSONEncoder to serialize to a JSON string.

Here is the property-by-property approach (do NOT use Mirror/reflection -- enumerate explicitly):

```swift
func readEffectiveConfig() -> EffectiveConfig {
    let userInfo = try? readConfig(scope: .user)
    let projectInfo = try? readConfig(scope: .project)
    let localInfo = try? readConfig(scope: .local)
    let managedInfo = readManagedConfig()

    let user = userInfo?.content ?? ClaudeConfig()
    let project = projectInfo?.content ?? ClaudeConfig()
    let local = localInfo?.content ?? ClaudeConfig()
    let managed = managedInfo?.content ?? ClaudeConfig()

    // Scopes in precedence order (lowest to highest)
    let scopes: [(ConfigScope, ClaudeConfig)] = [
        (.user, user), (.project, project), (.local, local), (.managed, managed)
    ]

    var merged = ClaudeConfig()
    var overrides: [ConfigOverride] = []

    // Helper to find winning scope for a key
    func resolve<T: Encodable>(
        key: String,
        _ extract: (ClaudeConfig) -> T?,
        _ assign: inout T?
    ) {
        let values: [(ConfigScope, T)] = scopes.compactMap { scope, cfg in
            extract(cfg).map { (scope, $0) }
        }
        guard let winner = values.last else { return }
        assign = winner.1

        // Serialize values to strings for the override
        let encoder = JSONEncoder()
        encoder.outputFormatting = .sortedKeys
        func serialize(_ val: T?) -> String? {
            guard let v = val else { return nil }
            if let s = v as? String { return s }
            if let b = v as? Bool { return b ? "true" : "false" }
            guard let data = try? encoder.encode(v),
                  let str = String(data: data, encoding: .utf8) else { return nil }
            return str
        }

        overrides.append(ConfigOverride(
            key: key,
            winningScope: winner.0,
            winningValue: serialize(winner.1) ?? "",
            userValue: serialize(extract(user)),
            projectValue: serialize(extract(project)),
            localValue: serialize(extract(local)),
            managedValue: serialize(extract(managed))
        ))
    }

    // Resolve each ClaudeConfig property
    resolve(key: "model", { $0.model }, &merged.model)
    resolve(key: "permissions", { $0.permissions }, &merged.permissions)
    resolve(key: "env", { $0.env }, &merged.env)
    resolve(key: "hooks", { $0.hooks }, &merged.hooks)
    resolve(key: "enabledPlugins", { $0.enabledPlugins }, &merged.enabledPlugins)
    resolve(key: "extraKnownMarketplaces", { $0.extraKnownMarketplaces }, &merged.extraKnownMarketplaces)
    resolve(key: "includeCoAuthoredBy", { $0.includeCoAuthoredBy }, &merged.includeCoAuthoredBy)
    resolve(key: "statusLine", { $0.statusLine }, &merged.statusLine)
    resolve(key: "alwaysThinkingEnabled", { $0.alwaysThinkingEnabled }, &merged.alwaysThinkingEnabled)
    resolve(key: "autoUpdatesChannel", { $0.autoUpdatesChannel }, &merged.autoUpdatesChannel)
    resolve(key: "theme", { $0.theme }, &merged.theme)
    resolve(key: "apiKeyStatus", { $0.apiKeyStatus }, &merged.apiKeyStatus)

    return EffectiveConfig(
        config: merged,
        overrides: overrides,
        profiles: ConfigProfiles(
            user: userInfo,
            project: projectInfo,
            local: localInfo,
            managed: managedInfo
        )
    )
}
```

NOTE: The generic `resolve` helper uses `inout T?` which requires the caller to pass `&merged.propertyName`. The Encodable constraint is needed for the JSONEncoder serialization. If the compiler complains about the generic, change the approach to use individual per-property blocks instead of a generic helper. The key requirement is: enumerate all 12 properties, find highest-precedence non-nil, build ConfigOverride.
