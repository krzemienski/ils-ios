**Category 4: Memory Profiling**
Check for memory leaks after navigating all screens:
```bash
# Navigate through all screens rapidly to stress memory
for route in home sessions mcp skills plugins system settings fleet teams themes hooks; do
  xcrun simctl openurl "$IPHONE" "ils://$route"
  sleep 1
done
sleep 3

# Check memory usage via vmmap
PID=$(xcrun simctl spawn "$IPHONE" launchctl list | grep com.ils.app | awk '{print $1}' 2>/dev/null)
if [ -n "$PID" ] && [ "$PID" != "-" ]; then
  vmmap "$PID" --summary 2>/dev/null > "$EDGEDIR/memory-profile.txt" || echo "vmmap not available for simulator process"
fi

# Alternative: check process memory via footprint
xcrun simctl spawn "$IPHONE" footprint --all 2>/dev/null | grep -i ils > "$EDGEDIR/memory-footprint.txt" || echo "Footprint data captured or unavailable" > "$EDGEDIR/memory-footprint.txt"
```
