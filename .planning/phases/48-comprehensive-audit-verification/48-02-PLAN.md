**Public endpoint audit script:**
```bash
BASE="http://localhost:9999"
OUTDIR="/tmp/v4.0-audit/backend"

# Health endpoints
for ep in "/health" "/health/ready" "/health/live"; do
  FNAME=$(echo "$ep" | tr '/' '-' | sed 's/^-//')
  curl -s "$BASE$ep" | python3 -c "
import json, sys
try:
    d = json.load(sys.stdin)
    print(json.dumps({'endpoint': '$ep', 'status': 'PASS', 'fields': list(d.keys())[:10]}, indent=2))
except Exception as e:
    print(json.dumps({'endpoint': '$ep', 'status': 'FAIL', 'error': str(e)}, indent=2))
" > "$OUTDIR/$FNAME.json" 2>&1
done

# API v1 endpoints (GET only for audit)
for ep in \
  "/api/v1/sessions" \
  "/api/v1/sessions/projects" \
  "/api/v1/skills" \
  "/api/v1/mcp" \
  "/api/v1/plugins" \
  "/api/v1/plugins/marketplace" \
  "/api/v1/stats" \
  "/api/v1/stats/recent" \
  "/api/v1/settings" \
  "/api/v1/server/status" \
  "/api/v1/themes" \
  "/api/v1/teams"; do
  FNAME=$(echo "$ep" | tr '/' '-' | sed 's/^-api-v1-//')
  curl -s "$BASE$ep" | python3 -c "
import json, sys
try:
    d = json.load(sys.stdin)
    has_wrapper = 'data' in d
    data = d.get('data', d)
    if isinstance(data, list) and len(data) > 0:
        fields = list(data[0].keys())[:10]
        count = len(data)
    elif isinstance(data, dict):
        fields = list(data.keys())[:10]
        count = 1
    else:
        fields = []
        count = 0
    # Check for snake_case (bad) vs camelCase (good)
    snake = [f for f in fields if '_' in f and not f.startswith('_')]
    print(json.dumps({
        'endpoint': '$ep',
        'status': 'PASS' if has_wrapper else 'FAIL',
        'wrapper': has_wrapper,
        'count': count,
        'fields': fields,
        'snake_case_fields': snake
    }, indent=2))
except Exception as e:
    print(json.dumps({'endpoint': '$ep', 'status': 'FAIL', 'error': str(e)}, indent=2))
" > "$OUTDIR/$FNAME.json" 2>&1
done
```
