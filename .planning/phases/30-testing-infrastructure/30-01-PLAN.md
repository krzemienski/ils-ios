**Step 1: Create `Tests/ILSSharedTests/TestDataFactories.swift`**

Create factory functions using Swift Testing framework conventions. Each factory returns a model instance with sensible defaults that can be overridden:

```swift
import Foundation
@testable import ILSShared

/// Test data factories for common ILSShared model objects.
/// Each factory provides sensible defaults; pass parameters to override specific fields.
enum TestFactory {
    static func makeSession(
        id: UUID = UUID(),
        name: String? = "Test Session",
        model: String = "sonnet",
        permissionMode: PermissionMode = .default,
        status: SessionStatus = .active,
        messageCount: Int = 5,
        source: SessionSource = .ils,
        createdAt: Date = Date(),
        lastActiveAt: Date = Date()
    ) -> ChatSession {
        ChatSession(
            id: id, name: name, model: model,
            permissionMode: permissionMode, status: status,
            messageCount: messageCount, source: source,
            createdAt: createdAt, lastActiveAt: lastActiveAt
        )
    }

    static func makeProject(
        id: UUID = UUID(),
        name: String = "Test Project",
        path: String = "/tmp/test-project",
        defaultModel: String = "sonnet"
    ) -> Project {
        Project(id: id, name: name, path: path, defaultModel: defaultModel)
    }

    static func makeMessage(
        id: UUID = UUID(),
        sessionId: UUID = UUID(),
        role: MessageRole = .user,
        content: String = "Hello, Claude"
    ) -> Message {
        Message(id: id, sessionId: sessionId, role: role, content: content)
    }

    static func makeExternalSession(
        claudeSessionId: String = "ext-session-001",
        name: String? = "External Session",
        projectPath: String? = "/tmp/ext-project",
        source: SessionSource = .external
    ) -> ExternalSession {
        ExternalSession(
            claudeSessionId: claudeSessionId, name: name,
            projectPath: projectPath, source: source
        )
    }
}
```
