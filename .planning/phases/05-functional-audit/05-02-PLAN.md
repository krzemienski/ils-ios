First, capture backend counts for correlation:
```bash

curl -s "http://localhost:9999/api/v1/mcp" | python3 -c "import sys,json; d=json.load(sys.stdin); print(json.dumps({'total': d.get('data',{}).get('total', len(d.get('data',{}).get('items',[])))}))" >> /tmp/ils-audit-evidence/phase5/faud06-curl-counts.json
# ... repeat for skills and plugins, or capture all in one script
```

Better approach -- single curl capture script:
```bash
python3 -c "
import json, urllib.request
counts = {}
for ep in ['mcp', 'skills', 'plugins']:
    resp = json.loads(urllib.request.urlopen(f'http://localhost:9999/api/v1/{ep}').read())
    data = resp.get('data', resp)
    if isinstance(data, dict):
        counts[ep] = {'total': data.get('total', len(data.get('items', [])))}
    elif isinstance(data, list):
        counts[ep] = {'total': len(data)}
with open('/tmp/ils-audit-evidence/phase5/faud06-curl-counts.json', 'w') as f:
    json.dump(counts, f, indent=2)
print(json.dumps(counts, indent=2))
"
```
